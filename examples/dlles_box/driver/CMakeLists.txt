cmake_minimum_required(VERSION 3.21)
project(DLLES_Driver CXX)
set(CMAKE_CXX_STANDARD 17)
list(APPEND CMAKE_PREFIX_PATH $ENV{LIBTORCH_DIR})
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
add_executable(dlles_driver driver.cpp)
target_link_libraries(dlles_driver "${TORCH_LIBRARIES}")
target_include_directories(dlles_driver PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
